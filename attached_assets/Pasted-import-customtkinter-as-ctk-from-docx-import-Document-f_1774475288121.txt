# --- MOTOR DE CÁLCULO (LÓGICA NBR 6118) ---
class EngineNBR6118:
    CAA_DATA = {
        "I (Fraca)": {"cobr": 20, "wk_lim": 0.4},
        "II (Moderada)": {"cobr": 25, "wk_lim": 0.3},
        "III (Forte)": {"cobr": 35, "wk_lim": 0.2},
        "IV (Muito Forte)": {"cobr": 45, "wk_lim": 0.2}
    }

    def __init__(self, fck, h_cm, caa, bitola_mm, espac_cm):
        self.fck = fck
        self.h = h_cm / 100
        self.caa = caa
        self.cobr = self.CAA_DATA[caa]["cobr"] / 1000
        self.phi = bitola_mm / 1000
        self.s = espac_cm / 100
        self.d = self.h - self.cobr - (self.phi / 2)
        self.As = (math.pi * (self.phi**2) / 4) * (1 / self.s) # m²/m
        
        # Constantes
        self.E_s = 210000000 # kN/m²
        self.alpha_e = 15 # Relação simplificada de módulos

    def calcular_estadio_ii(self, momento_k):
        b = 1.0
        A = 0.5 * b
        B = self.alpha_e * self.As
        C = -self.alpha_e * self.As * self.d
        x2 = (-B + math.sqrt(B**2 - 4*A*C)) / (2*A)
        I_ii = (b * x2**3 / 3) + self.alpha_e * self.As * (self.d - x2)**2
        sigma_s = self.alpha_e * (momento_k * (self.d - x2) / I_ii)
        return sigma_s, x2

    def verificar_tudo(self, m_elu, v_sd, m_els_w, m_fad_max, m_fad_min):
        # 1. Flexão ELU (Simplificado)
        fcd = (self.fck / 1.4) * 1000
        fyd = 434782 # kN/m² (CA-50)
        k_md = m_elu / (1.0 * (self.d**2) * fcd)
        limite_d3 = 0.45 # x/d limite
        
        # 2. Fissuração (wk)
        sig_w, _ = self.calcular_estadio_ii(m_els_w)
        wk = (self.phi / (12.5)) * (sig_w / self.E_s) * 3 * 1000 # mm
        wk_lim = self.CAA_DATA[self.caa]["wk_lim"]
        
        # 3. Fadiga
        sig_max, _ = self.calcular_estadio_ii(m_fad_max)
        sig_min, _ = self.calcular_estadio_ii(m_fad_min)
        delta_sig = abs(sig_max - sig_min)
        lim_fad = 190000 # kN/m² (Exemplo p/ 12.5mm)

        # 4. Cortante (VRd1)
        k = min(2.0, 1 + math.sqrt(200 / (self.d * 1000)))
        rho_l = min(0.02, self.As / (1.0 * self.d))
        v_rd1 = (0.12 * k * (100 * rho_l * self.fck)**(1/3)) * 1000 * 1.0 * self.d

        return {
            "wk": wk, "wk_lim": wk_lim, 
            "delta_sig": delta_sig, "lim_fad": lim_fad,
            "v_sd": v_sd, "v_rd1": v_rd1, "d_cm": self.d*100
        }
