import customtkinter as ctk
from docx import Document
import os

class BridgeSoftware(ctk.CTk):
    def __init__(self):
        super().__init__()

        self.title("BridgeDesign Pro - Módulo Inicial")
        self.geometry("600x700")
        ctk.set_appearance_mode("dark")

        # Variáveis de Armazenamento
        self.vao_y = ctk.StringVar(value="10.0")
        self.largura_x = ctk.StringVar(value="5.0")
        self.num_apoios = ctk.StringVar(value="2")
        self.tipo_viga = ctk.StringVar(value="Viga I (Pré-moldada)")
        self.tipo_laje = ctk.StringVar(value="Moldada in loco")

        self.setup_ui()

    def setup_ui(self):
        # Título
        self.label = ctk.CTkLabel(self, text="Definição do Empreendimento", font=("Arial", 20, "bold"))
        self.label.pack(pady=20)

        # Container Geometria
        self.geo_frame = ctk.CTkFrame(self)
        self.geo_frame.pack(pady=10, padx=20, fill="x")

        ctk.CTkLabel(self.geo_frame, text="GEOMETRIA INICIAL", font=("Arial", 14, "bold")).pack(pady=5)
        
        self.create_input(self.geo_frame, "Vão Longitudinal (Y) [m]:", self.vao_y)
        self.create_input(self.geo_frame, "Largura Total (X) [m]:", self.largura_x)
        self.create_input(self.geo_frame, "Número de Apoios:", self.num_apoios)

        # Container Superestrutura
        self.super_frame = ctk.CTkFrame(self)
        self.super_frame.pack(pady=10, padx=20, fill="x")

        ctk.CTkLabel(self.super_frame, text="TIPO DE SUPERESTRUTURA", font=("Arial", 14, "bold")).pack(pady=5)

        ctk.CTkLabel(self.super_frame, text="Tipo de Viga:").pack()
        self.viga_menu = ctk.CTkOptionMenu(self.super_frame, values=["Viga I (Pré-moldada)", "Viga T", "Viga Caixão"], variable=self.tipo_viga)
        self.viga_menu.pack(pady=5)

        ctk.CTkLabel(self.super_frame, text="Sistema da Laje:").pack()
        self.laje_menu = ctk.CTkOptionMenu(self.super_frame, values=["Moldada in loco", "Com Pré-lajes"], variable=self.tipo_laje)
        self.laje_menu.pack(pady=5)

        # Botão de Geração
        self.btn_gerar = ctk.CTkButton(self, text="Gerar Memória de Cálculo (.docx)", command=self.gerar_documento)
        self.btn_gerar.pack(pady=30)

    def create_input(self, frame, text, var):
        lbl = ctk.CTkLabel(frame, text=text)
        lbl.pack()
        entry = ctk.CTkEntry(frame, textvariable=var)
        entry.pack(pady=2)

    def gerar_documento(self):
        doc = Document()
        doc.add_heading('Memória de Cálculo Estrutural - Ponte', 0)

        doc.add_heading('1. Geometria e Definições', level=1)
        table = doc.add_table(rows=1, cols=2)
        table.style = 'Table Grid'
        
        data = [
            ("Vão Longitudinal (Y)", f"{self.vao_y.get()} m"),
            ("Largura Longitudinal (X)", f"{self.largura_x.get()} m"),
            ("Número de Apoios", self.num_apoios.get()),
            ("Tipo de Superestrutura", self.tipo_viga.get()),
            ("Tipo de Laje", self.tipo_laje.get())
        ]

        for item, value in data:
            row_cells = table.add_row().cells
            row_cells[0].text = item
            row_cells[1].text = value

        filename = "Memoria_Calculo_Ponte.docx"
        doc.save(filename)
        os.startfile(filename) # Abre o arquivo automaticamente no Windows

if __name__ == "__main__":
    app = BridgeSoftware()
    app.mainloop()
