import customtkinter as ctk
import math

class MotorEstrutural:
    @staticmethod
    def calcular_flexao_elu(m_sd, d, fck):
        if d <= 0 or abs(m_sd) == 0: return 0.0
        fcd = (fck / 1.4) / 10 
        fyd = 50 / 1.15 
        kmd = abs(m_sd) / (100 * (d**2) * fcd)
        if kmd > 0.372: return -1 
        kx = (1 - math.sqrt(1 - 2 * kmd)) / 0.8
        return abs(m_sd) / (fyd * d * (1 - 0.4 * kx))

    @staticmethod
    def calcular_as_min(fck, h):
        rho_min = max(0.0015, 0.035 * (fck/10)**(2/3) / 100)
        return rho_min * 100 * h

    @staticmethod
    def calcular_vrd_c(fck, d_cm, as_cm2, n_sd):
        """NBR 6118:2023 - Resistência ao cortante sem armadura transversal"""
        if d_cm <= 0: return 0.0
        d_m = d_cm / 100
        k = min(2.0, 1 + math.sqrt(200 / (d_m * 1000)))
        rho_l = min(0.02, as_cm2 / (100 * d_cm))
        # Tensão de compressão (Nsd positivo para compressão)
        sigma_cp = min(0.2 * (fck/1.4), abs(n_sd) / (100 * d_cm * 1.1)) 
        
        v_rdc = (0.12 * k * (100 * rho_l * fck)**(1/3) + 0.15 * sigma_cp) * 1000 * 1.0 * d_m
        v_min = (0.035 * k**(1.5) * fck**(0.5) + 0.15 * sigma_cp) * 1000 * 1.0 * d_m
        return max(v_rdc, v_min)
