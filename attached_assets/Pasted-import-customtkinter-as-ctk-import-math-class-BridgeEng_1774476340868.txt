import customtkinter as ctk
import math

class BridgeEngine:
    @staticmethod
    def calcular_vrd_c(fck, d_cm, as_cm2, n_sd=0):
        """
        Calcula a resistência ao cortante de lajes sem armadura transversal
        NBR 6118:2023 - Item 19.4.1
        """
        if d_cm <= 0: return 0.0
        
        # d em metros para as fórmulas de tensão
        d_m = d_cm / 100
        fck_mpa = fck
        
        # 1. Coeficiente k (Efeito de escala)
        k = min(2.0, 1 + math.sqrt(200 / (d_m * 1000)))
        
        # 2. Taxa de armadura longitudinal (rho_l) - limitada a 0.02
        rho_l = min(0.02, as_cm2 / (100 * d_cm))
        
        # 3. Tensão de compressão devida à força normal (sigma_cp)
        # Nsd em kN/m, Ac em cm² (100 * h)
        # Para este teste, simplificamos a área bruta Ac baseada na altura útil d
        sigma_cp = min(0.2 * (fck/1.4), abs(n_sd) / (100 * d_cm * 1.1)) # MPa
        
        # 4. Cálculo do VRd,c (em kN para uma faixa de 1 metro)
        # Fórmula: [0.12 * k * (100 * rho_l * fck)^(1/3) + 0.15 * sigma_cp] * b * d
        term1 = 0.12 * k * (100 * rho_l * fck_mpa)**(1/3)
        term2 = 0.15 * sigma_cp
        
        v_rdc = (term1 + term2) * 1000 * 1.0 * d_m # kN/m
        
        # Valor mínimo (Vmin)
        v_min = (0.035 * k**(1.5) * fck_mpa**(0.5) + 0.15 * sigma_cp) * 1000 * 1.0 * d_m
        
        return max(v_rdc, v_min)

class BridgeApp(ctk.CTk):
    def __init__(self):
        super().__init__()
        self.title("BridgeDesign Pro v5.0 - Verificação de Cortante NBR 6118")
        self.geometry("1300x950")

        # --- Variáveis de Cortante ---
        self.v_vsd = ctk.DoubleVar(value=150.0)    # Esforço do modelo
        self.v_nsd = ctk.DoubleVar(value=0.0)      # Força Normal (Compressão)
        self.v_h = ctk.DoubleVar(value=25.0)       # Espessura
        self.v_fck = ctk.DoubleVar(value=35.0)
        
        # Armadura longitudinal que contribui para o cortante
        self.v_as1 = ctk.DoubleVar(value=12.5)     # cm²/m (Vem do cálculo de flexão)

        self.setup_ui()

    def setup_ui(self):
        self.grid_columnconfigure(0, weight=1)
        self.grid_columnconfigure(1, weight=1)

        f_left = ctk.CTkScrollableFrame(self)
        f_left.grid(row=0, column=0, padx=15, pady=15, sticky="nsew")

        # Seção Cortante
        self.sect_title(f_left, "VERIFICAÇÃO DE CORTANTE (Vsd)")
        self.add_entry(f_left, "Vsd - Esforço Solicitante (kN/m):", self.v_vsd)
        self.add_entry(f_left, "Nsd - Força Normal (+ Comp) [kN/m]:", self.v_nsd)
        self.add_entry(f_left, "As1 - Armadura Longitudinal (cm²/m):", self.v_as1)
        self.add_entry(f_left, "Espessura da Laje (cm):", self.v_h)

        # Painel de Resultados
        self.f_res = ctk.CTkFrame(self, fg_color="#121212")
        self.f_res.grid(row=0, column=1, padx=15, pady=15, sticky="nsew")
        
        self.update()

    def sect_title(self, frame, text):
        ctk.CTkLabel(frame, text=text, font=("Arial", 14, "bold"), text_color="#1abc9c").pack(pady=15)

    def add_entry(self, frame, text, var):
        ctk.CTkLabel(frame, text=text).pack()
        e = ctk.CTkEntry(frame, textvariable=var)
        e.pack(pady=5)
        e.bind("<KeyRelease>", self.update)

    def update(self, *args):
        try:
            # d aproximado (cobrimento 2.5cm + bitola média)
            d_cm = self.v_h.get() - 2.5 - 0.8 
            
            v_rdc = BridgeEngine.calcular_vrd_c(
                self.v_fck.get(), 
                d_cm, 
                self.v_as1.get(), 
                self.v_nsd.get()
            )
            
            v_sd = self.v_vsd.get()
            status = v_sd <= v_rdc

            # Renderização
            for w in self.f_res.winfo_children(): w.destroy()
            
            ctk.CTkLabel(self.f_res, text="RESULTADO CORTANTE", font=("Arial", 18, "bold")).pack(pady=20)
            
            f = ctk.CTkFrame(self.f_res, fg_color="#1e1e1e", border_width=1)
            f.pack(padx=20, pady=10, fill="x")
            
            color = "#2ecc71" if status else "#e74c3c"
            ctk.CTkLabel(f, text=f"VRd,c (Resistência): {v_rdc:.2f} kN/m", font=("Arial", 16, "bold"), text_color="#3498db").pack(pady=10)
            ctk.CTkLabel(f, text=f"VSd (Solicitação): {v_sd:.2f} kN/m", font=("Arial", 14)).pack()
            
            ctk.CTkLabel(f, text="STATUS: ATENDE" if status else "STATUS: NÃO ATENDE", 
                         font=("Arial", 15, "bold"), text_color=color).pack(pady=15)
            
            if not status:
                ctk.CTkLabel(self.f_res, text="DICA: Aumente a espessura da laje\nou a armadura longitudinal (As1).", 
                             text_color="#f1c40f").pack()

        except: pass

if __name__ == "__main__":
    app = BridgeApp()
    app.mainloop()
